cmake_minimum_required(VERSION 3.22)
project(AnimalCrossingQuest C CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "Quest builds require the Android NDK toolchain.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(QUEST_WORKSPACE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(QUEST_SDL_ROOT "${QUEST_WORKSPACE}/third_party/SDL" CACHE PATH "Pinned SDL2 source")
set(QUEST_OPENXR_ROOT "${QUEST_WORKSPACE}/third_party/OpenXR-SDK" CACHE PATH "Pinned Khronos SDK")
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${QUEST_SDL_ROOT}" "sdl")
set(BUILD_LOADER ON CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
set(DYNAMIC_LOADER ON CACHE BOOL "" FORCE)
add_subdirectory("${QUEST_OPENXR_ROOT}" "openxr")
include_directories("${QUEST_OPENXR_ROOT}/include")
option(QUEST_BUILD_GAME "Build the full game instead of the XR bootstrap" ON)
if(QUEST_BUILD_GAME)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../pc" "game")
else()
    # Filled by the native OpenXR bootstrap workstream.
    include("${CMAKE_CURRENT_SOURCE_DIR}/probe.cmake")
endif()
